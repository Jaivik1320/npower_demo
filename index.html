Sequencing Engine — C# Core Logic
// OperationSequencer.cs — schedules ops earliest-start-first public class OperationSequencer { public IEnumerable<ScheduledOp> Sequence( IEnumerable<Operation> ops, ResourceCalendar calendar) { var queue = ops .OrderBy(o => o.Priority) .ThenBy(o => o.EarliestStart) .ToList(); foreach (var op in queue) { var slot = calendar.FindNextAvailable( op.ResourceId, op.EarliestStart, op.DurationHours); if (slot.HasConflict()) { yield return new ScheduledOp(op, slot, ScheduleStatus.Conflict); continue; } calendar.Reserve(op.ResourceId, slot); yield return new ScheduledOp(op, slot, ScheduleStatus.Scheduled); } } }

// Calculates operation completion date accounting // for shift hours, weekends, holidays public DateTime CalculateEndDate( DateTime start, decimal qty, decimal ratePerHour, ShiftCalendar shifts) { var totalHours = qty / ratePerHour; var remaining = totalHours; var cursor = start; while (remaining > 0) { var avail = shifts.HoursAvailable(cursor); if (avail > 0) remaining -= Math.Min(remaining, avail); cursor = cursor.AddDays(1); } return cursor; }
